# Clearly identifiable non-football or unreliable catch-all competitions.
EXCLUDED_LEAGUE_PATTERNS = [
    "%hockey%",
    "del %",
    "del2 %",
    "%darts%",
    "%esp8266%",
    "%spiele mit arminia bielefeld beteiligung%"
]


def get_connection():
    if not DATABASE.exists():
        raise FileNotFoundError(
            f"Database not found: {DATABASE}"
        )

    connection = sqlite3.connect(DATABASE)
    connection.row_factory = sqlite3.Row
    return connection


def exclusion_sql():
    return " AND ".join(
        ["LOWER(league) NOT LIKE ?" for _ in EXCLUDED_LEAGUE_PATTERNS]
    )


def exclusion_values():
    return tuple(
        pattern.lower()
        for pattern in EXCLUDED_LEAGUE_PATTERNS
    )

@app.get("/api/fixtures")
def get_fixtures(
    limit: int = Query(default=50, ge=1, le=100),
    source: str = Query(default="all")
):
    if source not in ("all", "football-data", "openligadb"):
        raise HTTPException(
            status_code=400,
            detail="Invalid source. Use all, football-data or openligadb."
        )

    source_condition = ""

    if source == "football-data":
        source_condition = " AND event_id LIKE 'football-data:%'"
    elif source == "openligadb":
        source_condition = " AND event_id LIKE 'openligadb:%'"

    with get_connection() as connection:

        total = connection.execute("""
            SELECT COUNT(*)
            FROM fixtures
            WHERE julianday(kickoff_utc) >= julianday('now')
        """).fetchone()[0]

        filtered_count = connection.execute(f"""
            SELECT COUNT(*)
            FROM fixtures
            WHERE julianday(kickoff_utc) >= julianday('now')
            {source_condition}
            AND {exclusion_sql()}
        """, exclusion_values()).fetchone()[0]

        rows = connection.execute(f"""
            SELECT
                event_id,
                league,
                home_team,
                away_team,
                kickoff_utc,
                kickoff_lagos,
                match_status,
                CASE
                    WHEN event_id LIKE 'football-data:%'
                        THEN 'football-data'
                    WHEN event_id LIKE 'openligadb:%'
                        THEN 'openligadb'
                    ELSE 'other'
                END AS source
            FROM fixtures
            WHERE julianday(kickoff_utc) >= julianday('now')
            {source_condition}
            AND {exclusion_sql()}
            ORDER BY julianday(kickoff_utc) ASC
            LIMIT ?
        """, (*exclusion_values(), limit)).fetchall()

    fixtures_list = [dict(row) for row in rows]

    return {
        "count": len(fixtures_list),
        "available_after_filter": filtered_count,
        "before_filter": total,
        "excluded": total - filtered_count,
        "source_filter": source,
        "fixtures": fixtures_list
    }


@app.get("/api/fixtures/{event_id}")
def fixture_details(event_id: str):

    with get_connection() as connection:
        row = connection.execute("""
            SELECT *
            FROM fixtures
            WHERE event_id = ?
        """, (event_id,)).fetchone()

    if row is None:
        raise HTTPException(
            status_code=404,
            detail="Fixture not found"
        )

    league = (row["league"] or "").lower()

    if any(
        pattern.replace("%", "") in league
        for pattern in EXCLUDED_LEAGUE_PATTERNS
    ):
        raise HTTPException(
            status_code=404,
            detail="This competition is excluded from the football API"
        )

    return dict(row)

function formatKickoff(fixture) {
  const value = fixture.kickoff_lagos || fixture.kickoff_utc
  if (!value) return 'Time unavailable'

  const date = new Date(value)
  if (Number.isNaN(date.getTime())) return value

  return new Intl.DateTimeFormat('en-NG', {
    timeZone: 'Africa/Lagos',
    weekday: 'short',
    day: 'numeric',
    month: 'short',
    hour: '2-digit',
    minute: '2-digit',
    hour12: true
  }).format(date)
}
